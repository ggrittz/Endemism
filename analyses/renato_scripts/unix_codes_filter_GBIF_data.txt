# Formattage de la liste dans un ficher temporaire (virer les “, la premiere ligne, puis les tabulations de fin de tableau, sinon ca marche pas), et stock le résultat dans un fichier temporaire ($$)
sed -e 's/"//g' treesSA.txt | awk 'NR !=1 {print $0}' | awk 'BEGIN {FS="\t"} {print $2}'  > $$ # the file need to be created using: sep="\t"
sed -e 's/"//g' treesSA.txt | awk 'NR !=1 {print $0}' | awk 'BEGIN {FS="\t"} {OFS=" "} {print $2}'  > $$

# Renomme le fichier temporaire
mv $$ treesSA_edited.txt
